Day 6 — OCPP message handlers

Goal: every message the charger sends updates the database correctly.

What each handler does

  • BootNotification → isOnline = true, reply Accepted + heartbeat interval.
  • Heartbeat → update lastSeenAt.
  • StatusNotification → save connector status. connectorId: 0 means the whole charger — ignore.
  • Authorize → accept only an idTag that belongs to a PENDING session.
  • StartTransaction → find the PENDING session by idTag, mark ACTIVE, return our transactionId.
  • MeterValues → save latest Wh reading, emit session:meter.
  • StopTransaction → bill the session (finishSession), emit session:stopped.
Always reply to StopTransaction, even if billing fails. Otherwise the charger retries forever.

import { prisma, Prisma } from "@ev/db";
import { events } from "../events";

export async function finishSession(transactionId: number, meterStopWh: number, stoppedAt: Date) {
  const session = await prisma.$transaction(async (tx) => {
    const s = await tx.chargingSession.findUnique({ where: { transactionId } });
    if (!s || s.status !== "ACTIVE") return null; // unknown or already billed → idempotent

    const usedWh = Math.max(0, meterStopWh - (s.meterStartWh ?? meterStopWh));
    const energyKwh = new Prisma.Decimal(usedWh).div(1000);
    const amount = energyKwh.mul(s.pricePerKwh).toDecimalPlaces(2);

    if (amount.gt(0)) {
      await tx.user.update({ where: { id: s.userId }, data: { walletBalance: { decrement: amount } } });
    }
    return tx.chargingSession.update({
      where: { id: s.id },
      data: { status: "COMPLETED", meterStopWh, meterLastWh: meterStopWh, energyKwh, amount, stoppedAt },
    });
  });
  if (session) events.emit("session:stopped", session);
}

import { prisma } from "@ev/db";
import { events } from "../events";
import { finishSession } from "../services/billing";

type Handler = (ocppId: string, payload: any) => Promise<object>;
const now = () => new Date().toISOString();

// Energy reading in Wh from a MeterValues payload (handles Wh or kWh units)
function readEnergyWh(meterValue: any[]): number | null {
  for (const mv of meterValue ?? []) {
    for (const sv of mv.sampledValue ?? []) {
      const isEnergy = !sv.measurand || sv.measurand === "Energy.Active.Import.Register";
      if (isEnergy) return Math.round(Number(sv.value) * (sv.unit === "kWh" ? 1000 : 1));
    }
  }
  return null;
}

export const handlers: Record<string, Handler> = {
  BootNotification: async (ocppId) => {
    await prisma.chargePoint.update({ where: { ocppId }, data: { isOnline: true, lastSeenAt: new Date() } });
    return { status: "Accepted", currentTime: now(), interval: 60 };
  },

  Heartbeat: async (ocppId) => {
    await prisma.chargePoint.update({ where: { ocppId }, data: { isOnline: true, lastSeenAt: new Date() } });
    return { currentTime: now() };
  },

  StatusNotification: async (ocppId, p) => {
    if (p.connectorId === 0) return {};
    const cp = await prisma.chargePoint.findUniqueOrThrow({ where: { ocppId } });
    await prisma.connector.upsert({
      where: { chargePointId_connectorId: { chargePointId: cp.id, connectorId: p.connectorId } },
      update: { status: p.status },
      create: { chargePointId: cp.id, connectorId: p.connectorId, status: p.status },
    });
    return {};
  },

  Authorize: async (_ocppId, p) => {
    const s = await prisma.chargingSession.findUnique({ where: { idTag: p.idTag } });
    return { idTagInfo: { status: s?.status === "PENDING" ? "Accepted" : "Invalid" } };
  },

  StartTransaction: async (_ocppId, p) => {
    const s = await prisma.chargingSession.findUnique({ where: { idTag: p.idTag } });
    if (!s || s.status !== "PENDING") return { transactionId: 0, idTagInfo: { status: "Invalid" } };

    const started = await prisma.chargingSession.update({
      where: { id: s.id },
      data: { status: "ACTIVE", meterStartWh: p.meterStart, meterLastWh: p.meterStart, startedAt: new Date(p.timestamp) },
    });
    events.emit("session:started", started);
    return { transactionId: started.transactionId, idTagInfo: { status: "Accepted" } };
  },

  MeterValues: async (_ocppId, p) => {
    const wh = readEnergyWh(p.meterValue);
    if (!p.transactionId || wh === null) return {};
    const s = await prisma.chargingSession.update({ where: { transactionId: p.transactionId }, data: { meterLastWh: wh } });
    events.emit("session:meter", s);
    return {};
  },

  StopTransaction: async (_ocppId, p) => {
    try {
      await finishSession(p.transactionId, p.meterStop, new Date(p.timestamp));
    } catch (err) {
      console.error("[billing] failed", p.transactionId, err); // fix by hand; never block the charger
    }
    return { idTagInfo: { status: "Accepted" } };
  },
};
