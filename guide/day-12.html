Day 12 — Super admin pages

Goal: we (super admin) can create tenants, create their admin logins, and register chargers.

Pages

  • /admin/tenants — list + "New tenant" form (name, slug, admin email, admin password)
  • /admin/chargers — list all chargers with tenant + online status, "Register charger" form (ocppId, name, tenant, number of connectors)

Tasks

apps/api/src/routes/admin.ts
import { Router } from "express";
import bcrypt from "bcryptjs";
import { z } from "zod";
import { prisma } from "@ev/db";
import { requireRole } from "../middleware/auth";

export const admin = Router();
admin.use(requireRole("SUPER_ADMIN"));

admin.get("/tenants", async (_req, res) => {
  res.json(await prisma.tenant.findMany({ include: { _count: { select: { chargePoints: true, memberships: true } } } }));
});

admin.post("/tenants", async (req, res) => {
  const body = z.object({
    name: z.string().min(2),
    slug: z.string().regex(/^[a-z0-9-]+$/),
    adminEmail: z.string().email(),
    adminPassword: z.string().min(8),
  }).parse(req.body);

  const tenant = await prisma.tenant.create({
    data: {
      name: body.name,
      slug: body.slug,
      admins: { create: { email: body.adminEmail, passwordHash: await bcrypt.hash(body.adminPassword, 10), role: "TENANT_ADMIN" } },
    },
  });
  res.status(201).json(tenant);
});

admin.get("/charge-points", async (_req, res) => {
  res.json(await prisma.chargePoint.findMany({ include: { tenant: true, connectors: true }, orderBy: { ocppId: "asc" } }));
});

admin.post("/charge-points", async (req, res) => {
  const body = z.object({
    ocppId: z.string().regex(/^[A-Za-z0-9-]+(_[A-Za-z0-9-]+)*$/).max(40),
    name: z.string().optional(),
    tenantId: z.string().uuid(),
    connectors: z.number().int().min(1).max(4),
  }).parse(req.body);

  const cp = await prisma.chargePoint.create({
    data: {
      ocppId: body.ocppId,
      name: body.name,
      tenantId: body.tenantId,
      allowedModes: [], // tenant admin turns modes on later
      connectors: { create: Array.from({ length: body.connectors }, (_, i) => ({ connectorId: i + 1 })) },
    },
  });
  res.status(201).json(cp);
});

Add an Express error handler at the end of index.ts that turns ZodError into 400 and Prisma P2002 (unique violation) into 409. You'll reuse it everywhere.
apps/api/src/index.ts — add last
import { ZodError } from "zod";

app.use((err: any, _req: express.Request, res: express.Response, _next: express.NextFunction) => {
  if (err instanceof ZodError) return void res.status(400).json(err.flatten());
  if (err?.code === "P2002") return void res.status(409).json({ error: "Already exists" });
  console.error(err);
  res.status(500).json({ error: "Internal error" });
});

Page pattern (use the same shape for both pages)

apps/web/src/app/admin/tenants/page.tsx
"use client";
import { useEffect, useState } from "react";
import { api } from "@/lib/api";

export default function Tenants() {
  const [rows, setRows] = useState<any[]>([]);
  const load = () => api("/admin/tenants").then(setRows);
  useEffect(() => { load(); }, []);

  async function create(form: FormData) {
    await api("/admin/tenants", { method: "POST", body: JSON.stringify(Object.fromEntries(form)) });
    load();
  }

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-semibold">Tenants</h1>
      <form action={create} className="grid grid-cols-5 gap-2">
        <input name="name" placeholder="Name" className="rounded border p-2" />
        <input name="slug" placeholder="slug" className="rounded border p-2" />
        <input name="adminEmail" placeholder="Admin email" className="rounded border p-2" />
        <input name="adminPassword" type="password" placeholder="Admin password" className="rounded border p-2" />
        <button className="rounded bg-black text-white">Create</button>
      </form>
      <table className="w-full text-left">
        <thead><tr><th>Name</th><th>Slug</th><th>Chargers</th><th>Members</th></tr></thead>
        <tbody>
          {rows.map((t) => (
            <tr key={t.id} className="border-t">
              <td>{t.name}</td><td>{t.slug}</td><td>{t._count.chargePoints}</td><td>{t._count.memberships}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

For /admin/chargers: same pattern. The form needs a <select name="tenantId"> filled from /admin/tenants, and send connectors as a number (Number(form.get("connectors"))).
