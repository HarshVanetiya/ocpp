Day 7 — Remote start / stop & sessions

Goal: startSession() and stopSession() that the Telegram bot will call. Test the full charge cycle today with a dev route — no bot yet.

The start sequence

flow
Bot: startSession()
 ├─ checkAccess() again           ← never trust an old button
 ├─ create session PENDING        ← locks mode + price, random idTag
 └─ RemoteStartTransaction ─────▶ Charger
                                   ├─ (Authorize idTag)
                                   └─ StartTransaction(idTag) ─▶ session ACTIVE, emit "session:started"

If the charger never sends StartTransaction (driver didn't plug in), mark the session FAILED after 2 minutes.

apps/api/src/services/session.ts
import { randomBytes } from "node:crypto";
import { prisma } from "@ev/db";
import { checkAccess } from "./access";
import { sendRemoteStart, sendRemoteStop } from "../ocpp/gateway";
import { events } from "../events";

const START_TIMEOUT_MS = 2 * 60_000;

export async function startSession(userId: string, ocppId: string, connectorNo: number, chatId?: string) {
  const { result, chargePoint, connector } = await checkAccess(userId, ocppId, connectorNo);
  if (!result.allowed) throw new Error(result.reason);

  const session = await prisma.chargingSession.create({
    data: {
      idTag: randomBytes(10).toString("hex"), // 20 chars = OCPP max
      userId,
      tenantId: chargePoint!.tenantId,
      connectorId: connector!.id,
      mode: result.mode,
      pricePerKwh: result.pricePerKwh,
      telegramChatId: chatId,
    },
  });

  const fail = async (reason: string) => {
    const s = await prisma.chargingSession.update({ where: { id: session.id }, data: { status: "FAILED" } });
    events.emit("session:failed", s, reason);
  };

  try {
    const res = await sendRemoteStart(ocppId, connectorNo, session.idTag);
    if (res.status !== "Accepted") { await fail("Charger rejected the start."); return session; }
  } catch (err) {
    await fail((err as Error).message);
    return session;
  }

  // Driver never plugged in?
  setTimeout(async () => {
    const s = await prisma.chargingSession.findUnique({ where: { id: session.id } });
    if (s?.status === "PENDING") await fail("No cable plugged in within 2 minutes.");
  }, START_TIMEOUT_MS);

  return session;
}

export async function stopSession(userId: string, sessionId: string) {
  const s = await prisma.chargingSession.findFirst({
    where: { id: sessionId, userId, status: "ACTIVE" },
    include: { connector: { include: { chargePoint: true } } },
  });
  if (!s) throw new Error("No active session found.");
  const res = await sendRemoteStop(s.connector.chargePoint.ocppId, s.transactionId);
  if (res.status !== "Accepted") throw new Error("Charger refused to stop. Unplug to end the session.");
  // Billing happens when the charger sends StopTransaction.
}
