Day 11 — Dashboard auth

Goal: admins log in, get a JWT in an httpOnly cookie, and every API route knows { adminId, role, tenantId }.

Design

  • Express owns auth. It checks the password, signs the JWT and enforces roles.
  • Next.js proxies /api/* → Express. Same origin = cookies just work, no CORS.
  • Tenant routes get req.db = tenantDb(tenantId). Isolation is automatic.

Tasks

apps/api/src/middleware/auth.ts
import type { RequestHandler } from "express";
import jwt from "jsonwebtoken";
import { tenantDb, type AdminRole, type TenantDb } from "@ev/db";
import { env } from "../env";

export type Claims = { adminId: string; role: AdminRole; tenantId: string | null };

declare global {
  namespace Express {
    interface Request { auth: Claims; db: TenantDb }
  }
}

export const requireRole = (...roles: AdminRole[]): RequestHandler => (req, res, next) => {
  try {
    const claims = jwt.verify(req.cookies?.token ?? "", env.JWT_SECRET) as Claims;
    if (!roles.includes(claims.role)) return void res.sendStatus(403);
    req.auth = claims;
    if (claims.role === "TENANT_ADMIN") req.db = tenantDb(claims.tenantId!);
    next();
  } catch {
    res.sendStatus(401);
  }
};
apps/api/src/routes/auth.ts
import { Router } from "express";
import bcrypt from "bcryptjs";
import jwt from "jsonwebtoken";
import { z } from "zod";
import { prisma } from "@ev/db";
import { env } from "../env";

export const auth = Router();

auth.post("/login", async (req, res) => {
  const { email, password } = z.object({ email: z.string().email(), password: z.string() }).parse(req.body);
  const admin = await prisma.adminUser.findUnique({ where: { email } });
  if (!admin || !(await bcrypt.compare(password, admin.passwordHash))) return void res.sendStatus(401);

  const claims = { adminId: admin.id, role: admin.role, tenantId: admin.tenantId };
  const token = jwt.sign(claims, env.JWT_SECRET, { expiresIn: "7d" });
  res.cookie("token", token, { httpOnly: true, sameSite: "lax", secure: process.env.NODE_ENV === "production", maxAge: 7 * 864e5 });
  res.json(claims);
});

auth.post("/logout", (_req, res) => { res.clearCookie("token").sendStatus(204); });

auth.get("/me", (req, res) => {
  try { res.json(jwt.verify(req.cookies?.token ?? "", env.JWT_SECRET)); }
  catch { res.sendStatus(401); }
});

apps/web/src/lib/api.ts
export async function api<T = any>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`/api${path}`, {
    ...init,
    headers: { "Content-Type": "application/json", ...init?.headers },
  });
  if (res.status === 401) { window.location.href = "/login"; throw new Error("Unauthorized"); }
  if (!res.ok) throw new Error(await res.text());
  return res.status === 204 ? (undefined as T) : res.json();
}
apps/web/src/app/login/page.tsx
"use client";
import { useState } from "react";
import { api } from "@/lib/api";

export default function Login() {
  const [error, setError] = useState("");

  async function submit(form: FormData) {
    try {
      const me = await api("/auth/login", { method: "POST", body: JSON.stringify(Object.fromEntries(form)) });
      window.location.href = me.role === "SUPER_ADMIN" ? "/admin/tenants" : "/dashboard/chargers";
    } catch { setError("Wrong email or password"); }
  }

  return (
    <form action={submit} className="mx-auto mt-24 flex max-w-sm flex-col gap-3">
      <h1 className="text-2xl font-semibold">Sign in</h1>
      <input name="email" type="email" placeholder="Email" className="rounded border p-2" required />
      <input name="password" type="password" placeholder="Password" className="rounded border p-2" required />
      <button className="rounded bg-black p-2 text-white">Login</button>
      {error && <p className="text-red-600">{error}</p>}
    </form>
  );
}

Protect each area with a client layout that checks /auth/me:

apps/web/src/components/RequireRole.tsx
"use client";
import { useEffect, useState, type ReactNode } from "react";
import { api } from "@/lib/api";

export function RequireRole({ role, children }: { role: "SUPER_ADMIN" | "TENANT_ADMIN"; children: ReactNode }) {
  const [ok, setOk] = useState(false);
  useEffect(() => {
    api("/auth/me").then((me) => (me.role === role ? setOk(true) : (window.location.href = "/login")));
  }, [role]);
  return ok ? <>{children}</> : null;
}
