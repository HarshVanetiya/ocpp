Day 9 — QR scan & charging flow

Goal: scan → see mode & price → ⚡ Start → live status message updating → 🛑 Stop → receipt.

QR payload format

https://t.me/<BOT_USERNAME>?start=cp_<ocppId>_conn_<connectorNo>

Telegram allows only A–Z a–z 0–9 _ - and max 64 chars, so keep ocppId simple. Since ocppId may itself contain _, parse with a regex anchored on _conn_<digits> at the end.

apps/api/src/bot/bot.ts — replace the handleScan placeholder
import { InlineKeyboard } from "grammy";
import { checkAccess } from "../services/access";
import { startSession, stopSession } from "../services/session";

const SCAN_RE = /^cp_(.+)_conn_(\d+)$/;
const MODE_LABEL = { FLEET: "🚚 Fleet mode", PRIVATE_MEMBER: "🏠 Member mode", PUBLIC: "🌐 Public mode" } as const;

export async function handleScan(ctx: Context, user: User, payload: string) {
  const m = payload.match(SCAN_RE);
  if (!m) return ctx.reply("❓ This QR code is not valid.");
  const [, ocppId, conn] = m;

  const { result, chargePoint, connector } = await checkAccess(user.id, ocppId, Number(conn));
  if (!result.allowed) return ctx.reply(result.reason);

  const price = result.pricePerKwh.isZero() ? "Free" : `₹${result.pricePerKwh}/kWh`;
  await ctx.reply(
    [
      `🔌 ${chargePoint!.name ?? ocppId} · Connector ${conn}`,
      `Status: ${connector!.status}`,
      `${MODE_LABEL[result.mode]} · ${price}`,
      `💰 Wallet: ₹${user.walletBalance}`,
    ].join("\n"),
    { reply_markup: new InlineKeyboard().text(`⚡ Start Charging (${price})`, `start:${ocppId}:${conn}`) },
  );
}

bot.callbackQuery(/^start:(.+):(\d+)$/, async (ctx) => {
  await ctx.answerCallbackQuery();
  await ctx.editMessageReplyMarkup(); // remove the button → no double taps
  const user = await getUser(ctx);
  try {
    await startSession(user.id, ctx.match[1], Number(ctx.match[2]), String(ctx.chat!.id));
    await ctx.reply("⏳ Starting… plug in your cable now.");
  } catch (err) {
    await ctx.reply(`❌ ${(err as Error).message}`);
  }
});

bot.callbackQuery(/^stop:(.+)$/, async (ctx) => {
  await ctx.answerCallbackQuery("Stopping…");
  const user = await getUser(ctx);
  try {
    await stopSession(user.id, ctx.match[1]);
  } catch (err) {
    await ctx.reply(`❌ ${(err as Error).message}`);
  }
});

Live status: one message, edited

On session:started we send one message, pin it, and save its message_id. Every MeterValues edits that same message (throttled to once per 15 s). On stop we unpin it and send a receipt.

apps/api/src/bot/notify.ts
import { InlineKeyboard } from "grammy";
import { prisma, type ChargingSession } from "@ev/db";
import { bot } from "./bot";
import { events } from "../events";

const lastEdit = new Map<string, number>();
const stopKb = (id: string) => new InlineKeyboard().text("🛑 Stop Charging", `stop:${id}`);

function liveText(s: ChargingSession) {
  const kwh = ((s.meterLastWh ?? 0) - (s.meterStartWh ?? 0)) / 1000;
  const cost = s.pricePerKwh.mul(kwh).toFixed(2);
  return `⚡ Charging (${s.mode})\nEnergy: ${kwh.toFixed(2)} kWh\nCost so far: ₹${cost}\nUpdated: ${new Date().toLocaleTimeString("en-IN")}`;
}

events.on("session:started", async (s) => {
  if (!s.telegramChatId) return;
  const msg = await bot.api.sendMessage(s.telegramChatId, liveText(s), { reply_markup: stopKb(s.id) });
  await prisma.chargingSession.update({ where: { id: s.id }, data: { statusMessageId: msg.message_id } });
  await bot.api.pinChatMessage(s.telegramChatId, msg.message_id, { disable_notification: true }).catch(() => {});
});

events.on("session:meter", async (s) => {
  if (!s.telegramChatId || !s.statusMessageId) return;
  if (Date.now() - (lastEdit.get(s.id) ?? 0) < 15_000) return;
  lastEdit.set(s.id, Date.now());
  await bot.api
    .editMessageText(s.telegramChatId, s.statusMessageId, liveText(s), { reply_markup: stopKb(s.id) })
    .catch(() => {}); // "message is not modified" is fine
});

events.on("session:stopped", async (s) => {
  lastEdit.delete(s.id);
  if (!s.telegramChatId) return;
  if (s.statusMessageId) {
    await bot.api.unpinChatMessage(s.telegramChatId, s.statusMessageId).catch(() => {});
    await bot.api.editMessageReplyMarkup(s.telegramChatId, s.statusMessageId).catch(() => {});
  }
  const user = await prisma.user.findUniqueOrThrow({ where: { id: s.userId } });
  await bot.api.sendMessage(
    s.telegramChatId,
    `🧾 Session complete\nEnergy: ${s.energyKwh} kWh\nRate: ₹${s.pricePerKwh}/kWh (${s.mode})\nAmount: ₹${s.amount}\nWallet balance: ₹${user.walletBalance}`,
  );
});

events.on("session:failed", async (s, reason) => {
  if (s.telegramChatId) await bot.api.sendMessage(s.telegramChatId, `❌ Could not start: ${reason}`);
});
