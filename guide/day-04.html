Day 4 — Access & tariff engine

Goal: given (userId, ocppId, connectorNo) decide: allowed or not, which mode, which price.

Split it in two

  • decideAccess(input) — pure function, no database. Easy to unit test.
  • checkAccess(userId, ocppId, connectorNo) — loads data from Prisma, then calls decideAccess.

The waterfall

  1. Charger exists, is online, connector is Available or Preparing. User has no other running session.
  2. Case A: user has an active membership in the charger's tenant and that role is in allowedModes and a tariff rule exists → use it.
  3. Case B: else, if PUBLIC is in allowedModes → public tariff.
  4. Else → ⛔ This charger is restricted to authorized Fleet/Society members only.
  5. Any mode with price > 0 → wallet must be ≥ MIN_WALLET_BALANCE.

apps/api/src/services/access.ts
import { prisma, Prisma, type AccessMode } from "@ev/db";
import { env } from "../env";

export type AccessResult =
  | { allowed: true; mode: AccessMode; pricePerKwh: Prisma.Decimal }
  | { allowed: false; reason: string };

export type AccessInput = {
  online: boolean;
  connectorStatus: string;
  hasRunningSession: boolean;
  allowedModes: AccessMode[];
  tariffs: { mode: AccessMode; pricePerKwh: Prisma.Decimal }[];
  memberRole: AccessMode | null; // active membership in THIS charger's tenant
  walletBalance: Prisma.Decimal;
  minBalance: number;
};

const deny = (reason: string): AccessResult => ({ allowed: false, reason });

export function decideAccess(i: AccessInput): AccessResult {
  if (!i.online) return deny("📴 This charger is offline right now.");
  if (!["Available", "Preparing"].includes(i.connectorStatus))
    return deny(`🔌 Connector is ${i.connectorStatus}. Try another one.`);
  if (i.hasRunningSession) return deny("⚡ You already have a charging session running.");

  const tariff = (m: AccessMode) => i.tariffs.find((t) => t.mode === m);
  let pick: { mode: AccessMode; pricePerKwh: Prisma.Decimal } | undefined;

  // Case A — member of this tenant with an allowed role
  if (i.memberRole && i.allowedModes.includes(i.memberRole)) pick = tariff(i.memberRole);

  // Case B — fall back to public
  if (!pick && i.allowedModes.includes("PUBLIC")) {
    pick = tariff("PUBLIC");
    if (!pick) return deny("⚙️ Charger is not configured yet. Contact the operator.");
  }

  if (!pick) return deny("⛔ This charger is restricted to authorized Fleet/Society members only.");

  if (pick.pricePerKwh.gt(0) && i.walletBalance.lt(i.minBalance))
    return deny(`💰 You need at least ₹${i.minBalance} in your wallet. Use /topup.`);

  return { allowed: true, ...pick };
}

export async function checkAccess(userId: string, ocppId: string, connectorNo: number) {
  const cp = await prisma.chargePoint.findUnique({
    where: { ocppId },
    include: { tariffRules: true, connectors: { where: { connectorId: connectorNo } } },
  });
  const connector = cp?.connectors[0];
  if (!cp || !connector) return { result: deny("❓ Unknown charger or connector.") };

  const user = await prisma.user.findUniqueOrThrow({
    where: { id: userId },
    include: { memberships: { where: { tenantId: cp.tenantId, isActive: true } } },
  });
  const running = await prisma.chargingSession.count({
    where: { userId, status: { in: ["PENDING", "ACTIVE"] } },
  });

  const result = decideAccess({
    online: cp.isOnline,
    connectorStatus: connector.status,
    hasRunningSession: running > 0,
    allowedModes: cp.allowedModes,
    tariffs: cp.tariffRules,
    memberRole: user.memberships[0]?.role ?? null,
    walletBalance: user.walletBalance,
    minBalance: env.MIN_WALLET_BALANCE,
  });
  return { result, chargePoint: cp, connector, user };
}

Tests

apps/api/src/services/access.test.ts
import { describe, it, expect } from "vitest";
import { Prisma } from "@ev/db";
import { decideAccess, type AccessInput } from "./access";

const D = (n: number) => new Prisma.Decimal(n);
const base: AccessInput = {
  online: true, connectorStatus: "Available", hasRunningSession: false,
  allowedModes: ["FLEET", "PUBLIC"],
  tariffs: [{ mode: "FLEET", pricePerKwh: D(0) }, { mode: "PUBLIC", pricePerKwh: D(18) }],
  memberRole: null, walletBalance: D(0), minBalance: 100,
};

describe("decideAccess", () => {
  it("fleet member charges free", () => {
    const r = decideAccess({ ...base, memberRole: "FLEET" });
    expect(r).toMatchObject({ allowed: true, mode: "FLEET" });
  });
  it("stranger becomes PUBLIC", () => {
    const r = decideAccess({ ...base, walletBalance: D(500) });
    expect(r).toMatchObject({ allowed: true, mode: "PUBLIC" });
  });
  it("public with low wallet is denied", () => {
    expect(decideAccess(base).allowed).toBe(false);
  });
  it("member with disallowed role falls back to PUBLIC", () => {
    const r = decideAccess({ ...base, memberRole: "PRIVATE_MEMBER", walletBalance: D(500) });
    expect(r).toMatchObject({ allowed: true, mode: "PUBLIC" });
  });
  it("fleet-only charger rejects strangers", () => {
    const r = decideAccess({ ...base, allowedModes: ["FLEET"] });
    expect(r.allowed).toBe(false);
  });
  it("offline charger is denied", () => {
    expect(decideAccess({ ...base, memberRole: "FLEET", online: false }).allowed).toBe(false);
  });
});
