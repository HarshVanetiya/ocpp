Day 5 — Express + OCPP WebSocket

Goal: chargers connect to ws://localhost:4000/ocpp/:ocppId, we keep their sockets in a Map, and we can both answer and send OCPP calls.

OCPP 1.6J in 60 seconds

Every message is a JSON array. Only three shapes exist:

message types
CALL        [2, "<uniqueId>", "<Action>", {payload}]     ← request (either side)
CALLRESULT  [3, "<uniqueId>", {payload}]                 ← success reply
CALLERROR   [4, "<uniqueId>", "<ErrorCode>", "<description>", {}]
  • Charger → us: BootNotification, Heartbeat, StatusNotification, Authorize, StartTransaction, MeterValues, StopTransaction
  • Us → charger: RemoteStartTransaction, RemoteStopTransaction
  • The WebSocket subprotocol must be ocpp1.6, or real chargers will disconnect.
  • Reply to a CALL with the same uniqueId.

apps/api/src/ocpp/gateway.ts
import { WebSocketServer, WebSocket } from "ws";
import type { Server } from "node:http";
import { randomUUID } from "node:crypto";
import { prisma } from "@ev/db";
import { handlers } from "./handlers";

export const sockets = new Map<string, WebSocket>();
const pending = new Map<string, { resolve: (p: any) => void; reject: (e: Error) => void; timer: NodeJS.Timeout }>();

export function attachOcpp(server: Server) {
  const wss = new WebSocketServer({
    noServer: true,
    handleProtocols: (protocols) => (protocols.has("ocpp1.6") ? "ocpp1.6" : false),
  });

  server.on("upgrade", async (req, socket, head) => {
    const m = req.url?.match(/^\/ocpp\/([^/?]+)/);
    const ocppId = m && decodeURIComponent(m[1]);
    const known = ocppId && (await prisma.chargePoint.findUnique({ where: { ocppId } }));
    if (!known) {
      socket.write("HTTP/1.1 404 Not Found\r\n\r\n");
      return socket.destroy();
    }
    wss.handleUpgrade(req, socket, head, (ws) => onConnect(ws, ocppId));
  });
}

function onConnect(ws: WebSocket, ocppId: string) {
  sockets.get(ocppId)?.terminate(); // charger reconnected → drop the stale socket
  sockets.set(ocppId, ws);
  console.log(`[ocpp] ${ocppId} connected`);

  ws.on("message", async (raw) => {
    let msg: any[];
    try { msg = JSON.parse(raw.toString()); } catch { return; }
    const [type, id] = msg;

    if (type === 2) {
      const [, , action, payload] = msg;
      console.log(`[ocpp] ${ocppId} → ${action}`, payload);
      const handler = handlers[action];
      if (!handler) return ws.send(JSON.stringify([4, id, "NotImplemented", action, {}]));
      try {
        ws.send(JSON.stringify([3, id, await handler(ocppId, payload)]));
      } catch (err) {
        console.error(err);
        ws.send(JSON.stringify([4, id, "InternalError", String(err), {}]));
      }
      return;
    }

    // Reply to a call WE made
    const p = pending.get(id);
    if (!p) return;
    clearTimeout(p.timer);
    pending.delete(id);
    if (type === 3) p.resolve(msg[2]);
    else p.reject(new Error(`${msg[2]}: ${msg[3]}`));
  });

  ws.on("close", async () => {
    if (sockets.get(ocppId) !== ws) return; // already replaced
    sockets.delete(ocppId);
    await prisma.chargePoint.update({ where: { ocppId }, data: { isOnline: false } });
    console.log(`[ocpp] ${ocppId} disconnected`);
  });
}

// Send a CALL to a charger and wait for its reply.
export function call<T = any>(ocppId: string, action: string, payload: object, timeoutMs = 30_000): Promise<T> {
  const ws = sockets.get(ocppId);
  if (!ws || ws.readyState !== WebSocket.OPEN) return Promise.reject(new Error("Charger is offline"));
  const id = randomUUID();
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => { pending.delete(id); reject(new Error(`${action} timed out`)); }, timeoutMs);
    pending.set(id, { resolve, reject, timer });
    ws.send(JSON.stringify([2, id, action, payload]));
  });
}

apps/api/scripts/sim.ts
import WebSocket from "ws";

const ID = process.argv[2] ?? "CP_TEST01";
const ws = new WebSocket(`ws://localhost:4000/ocpp/${ID}`, "ocpp1.6");
const waiting = new Map<string, (p: any) => void>();
let seq = 0, meter = 10_000, tx: number | null = null;
let timer: NodeJS.Timeout | undefined;

const call = (action: string, payload: object) =>
  new Promise<any>((res) => {
    const id = String(++seq);
    waiting.set(id, res);
    ws.send(JSON.stringify([2, id, action, payload]));
  });
const status = (connectorId: number, status: string) =>
  call("StatusNotification", { connectorId, errorCode: "NoError", status });
const now = () => new Date().toISOString();

ws.on("open", async () => {
  console.log("Boot:", await call("BootNotification", { chargePointVendor: "Sim", chargePointModel: "v1" }));
  await status(1, "Available");
  await status(2, "Available");
  setInterval(() => call("Heartbeat", {}), 60_000);
});

ws.on("message", async (raw) => {
  const msg = JSON.parse(raw.toString());
  if (msg[0] === 3 || msg[0] === 4) { waiting.get(msg[1])?.(msg[2]); waiting.delete(msg[1]); return; }
  if (msg[0] !== 2) return;
  const [, id, action, p] = msg;
  console.log("<<", action, p);

  if (action === "RemoteStartTransaction") {
    ws.send(JSON.stringify([3, id, { status: "Accepted" }]));
    await status(p.connectorId, "Preparing");
    const res = await call("StartTransaction", { connectorId: p.connectorId, idTag: p.idTag, meterStart: meter, timestamp: now() });
    if (res?.idTagInfo?.status !== "Accepted") return status(p.connectorId, "Available");
    tx = res.transactionId;
    await status(p.connectorId, "Charging");
    timer = setInterval(() => {
      meter += 300; // +0.3 kWh every 5 s
      call("MeterValues", {
        connectorId: p.connectorId, transactionId: tx,
        meterValue: [{ timestamp: now(), sampledValue: [{ value: String(meter), measurand: "Energy.Active.Import.Register", unit: "Wh" }] }],
      });
    }, 5_000);
  } else if (action === "RemoteStopTransaction") {
    ws.send(JSON.stringify([3, id, { status: "Accepted" }]));
    clearInterval(timer);
    await call("StopTransaction", { transactionId: tx, meterStop: meter, timestamp: now(), reason: "Remote" });
    await status(1, "Available");
  } else {
    ws.send(JSON.stringify([4, id, "NotImplemented", "", {}]));
  }
});
