Day 8 — Telegram bot & onboarding

Goal: a new user opens the bot, shares their phone, and gets auto-linked to any fleet/society that already added that number.

Why phone sharing?

Admins only know a driver's phone number. Telegram's request_contact button gives us a verified number. We match it against TenantMembership.phone and fill in userId.

apps/api/src/bot/bot.ts
import { Bot, Keyboard, type Context } from "grammy";
import { prisma, normalizePhone, type User } from "@ev/db";
import { env } from "../env";

export const bot = new Bot(env.TELEGRAM_BOT_TOKEN!);

// QR payload remembered while we wait for the phone number
const pendingScan = new Map<string, string>();

const askPhone = new Keyboard().requestContact("📱 Share my phone number").resized().oneTime();

export async function getUser(ctx: Context): Promise<User> {
  const telegramId = String(ctx.from!.id);
  return prisma.user.upsert({
    where: { telegramId },
    update: {},
    create: { telegramId, name: ctx.from!.first_name },
  });
}

bot.command("start", async (ctx) => {
  const user = await getUser(ctx);
  const payload = ctx.match; // "cp_CP_TEST01_conn_1" or ""

  if (!user.phone) {
    if (payload) pendingScan.set(user.id, payload);
    return ctx.reply("👋 Welcome! Share your phone number to continue.", { reply_markup: askPhone });
  }
  if (payload) return handleScan(ctx, user, payload);
  return ctx.reply("Scan the QR code on a charger to start charging ⚡");
});

bot.on("message:contact", async (ctx) => {
  const contact = ctx.message.contact;
  if (contact.user_id !== ctx.from.id) {
    return ctx.reply("Please share your own number using the button.", { reply_markup: askPhone });
  }

  const phone = normalizePhone(contact.phone_number);
  const user = await prisma.user.update({ where: { telegramId: String(ctx.from.id) }, data: { phone } });
  const linked = await prisma.tenantMembership.updateMany({ where: { phone, userId: null }, data: { userId: user.id } });

  await ctx.reply(
    `✅ Verified${linked.count ? ` — linked to ${linked.count} organisation(s)` : ""}.`,
    { reply_markup: { remove_keyboard: true } },
  );

  const payload = pendingScan.get(user.id);
  if (payload) {
    pendingScan.delete(user.id);
    return handleScan(ctx, user, payload);
  }
});

bot.command("wallet", async (ctx) => {
  const user = await getUser(ctx);
  await ctx.reply(`💰 Wallet balance: ₹${user.walletBalance}`);
});

bot.catch((err) => console.error("[bot]", err.error));

// Filled in on Day 9
export async function handleScan(ctx: Context, user: User, payload: string) {
  await ctx.reply(`Scanned: ${payload}`);
}
