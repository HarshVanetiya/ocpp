Day 2 — Database schema & seed

Goal: the full Prisma schema, first migration, and seed data you can test with all week.

What changed from the original plan (and why)

  • Added ChargingSession — it was referenced but never defined.
  • TenantMembership.phone + optional userId — admins add drivers by phone before the driver has ever opened the bot. We link userId when they share their contact.
  • Session has transactionId (Int) and idTag (≤20 chars) — OCPP 1.6 requires both in these formats.
  • Session stores mode + pricePerKwh — the price is locked at start.
  • Added AdminUser (dashboard logins) and Payment (wallet top-ups).
  • Money is Decimal(12,2), never Float.

The schema

Red squiggle on directUrl in VS Code? The Prisma VS Code extension may say "directUrl is no longer supported in schema files, move it to prisma.config.ts". That rule is for Prisma 7. We pin Prisma 6, where url and directUrl belong in the schema. Run pnpm exec prisma validate in packages/db: if it says "valid", ignore the squiggle. Do not move the URLs to a config file. Fix: keep only the stable Prisma extension (disable Prisma - Insider), then in VS Code settings (Ctrl+,, search prisma) turn on Pin to Prisma 6 and reload the window.
packages/db/prisma/schema.prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url       = env("DATABASE_URL") // pooled: used by the running app
  directUrl = env("DIRECT_URL")   // direct: used by prisma migrate
}

enum AccessMode {
  PUBLIC
  PRIVATE_MEMBER // society resident / office employee
  FLEET          // commercial fleet driver
}

enum AdminRole {
  SUPER_ADMIN
  TENANT_ADMIN
}

enum SessionStatus {
  PENDING   // remote start sent, waiting for StartTransaction
  ACTIVE
  COMPLETED
  FAILED
}

enum PaymentStatus {
  CREATED
  PAID
  FAILED
}

model Tenant {
  id           String              @id @default(uuid())
  name         String
  slug         String              @unique
  country      String              @default("IN") // ISO code (IN, GB, AE...): default for local phone numbers
  createdAt    DateTime            @default(now())
  chargePoints ChargePoint[]
  memberships  TenantMembership[]
  sessions     ChargingSession[]
  tariffRules  ChargerTariffRule[]
  admins       AdminUser[]
}

model User {
  id            String             @id @default(uuid())
  telegramId    String             @unique
  phone         String?            @unique // E.164, e.g. +919876543210
  name          String?
  walletBalance Decimal            @default(0) @db.Decimal(12, 2)
  createdAt     DateTime           @default(now())
  memberships   TenantMembership[]
  sessions      ChargingSession[]
  payments      Payment[]
}

model TenantMembership {
  id        String     @id @default(uuid())
  tenantId  String
  phone     String     // what the admin types in
  userId    String?    // filled when the driver shares their contact
  role      AccessMode // FLEET or PRIVATE_MEMBER only
  name      String?
  vehicleNo String?
  isActive  Boolean    @default(true)
  tenant    Tenant     @relation(fields: [tenantId], references: [id])
  user      User?      @relation(fields: [userId], references: [id])

  @@unique([tenantId, phone])
  @@unique([tenantId, userId])
}

model ChargePoint {
  id           String              @id @default(uuid())
  ocppId       String              @unique // "CP_101"
  name         String?
  tenantId     String
  allowedModes AccessMode[]        // ["FLEET"], ["FLEET","PUBLIC"], ...
  isOnline     Boolean             @default(false)
  lastSeenAt   DateTime?
  tenant       Tenant              @relation(fields: [tenantId], references: [id])
  connectors   Connector[]
  tariffRules  ChargerTariffRule[]
}

model Connector {
  id            String            @id @default(uuid())
  chargePointId String
  connectorId   Int               // 1, 2 (OCPP number)
  status        String            @default("Unavailable")
  chargePoint   ChargePoint       @relation(fields: [chargePointId], references: [id])
  sessions      ChargingSession[]

  @@unique([chargePointId, connectorId])
}

model ChargerTariffRule {
  id            String      @id @default(uuid())
  chargePointId String
  tenantId      String
  mode          AccessMode
  pricePerKwh   Decimal     @db.Decimal(10, 2)
  chargePoint   ChargePoint @relation(fields: [chargePointId], references: [id])
  tenant        Tenant      @relation(fields: [tenantId], references: [id])

  @@unique([chargePointId, mode])
}

model ChargingSession {
  id              String        @id @default(uuid())
  transactionId   Int           @unique @default(autoincrement()) // sent to charger
  idTag           String        @unique // random, ≤ 20 chars
  userId          String
  tenantId        String
  connectorId     String
  mode            AccessMode    // locked at start
  pricePerKwh     Decimal       @db.Decimal(10, 2) // locked at start
  status          SessionStatus @default(PENDING)
  meterStartWh    Int?
  meterLastWh     Int?
  meterStopWh     Int?
  energyKwh       Decimal?      @db.Decimal(10, 3)
  amount          Decimal?      @db.Decimal(12, 2)
  telegramChatId  String?
  statusMessageId Int?          // the live message we keep editing
  createdAt       DateTime      @default(now())
  startedAt       DateTime?
  stoppedAt       DateTime?
  user            User          @relation(fields: [userId], references: [id])
  tenant          Tenant        @relation(fields: [tenantId], references: [id])
  connector       Connector     @relation(fields: [connectorId], references: [id])

  @@index([tenantId, createdAt])
  @@index([userId, status])
}

model AdminUser {
  id           String    @id @default(uuid())
  email        String    @unique
  passwordHash String
  role         AdminRole
  tenantId     String?   // null for SUPER_ADMIN
  tenant       Tenant?   @relation(fields: [tenantId], references: [id])
}

model Payment {
  id             String        @id @default(uuid())
  userId         String
  amount         Decimal       @db.Decimal(12, 2)
  status         PaymentStatus @default(CREATED)
  razorpayLinkId String?       @unique
  createdAt      DateTime      @default(now())
  paidAt         DateTime?
  user           User          @relation(fields: [userId], references: [id])
}

Phones must work for any country, not just India. Country codes are 1–3 digits and every country has its own local format, so don't hand-roll this: use libphonenumber-js (a port of Google's libphonenumber).

pnpm --filter @ev/db add libphonenumber-js
packages/db/src/phone.ts
import { parsePhoneNumberFromString, type CountryCode } from "libphonenumber-js";

// One phone format everywhere: E.164, e.g. "+919876543210", "+447911123456".
// defaultCountry is only used when the input has no "+<country code>" (an admin typing a local number).
export function normalizePhone(input: string, defaultCountry: CountryCode): string {
  const phone = parsePhoneNumberFromString(input, defaultCountry);
  if (!phone || !phone.isValid()) throw new Error(`Invalid phone number: ${input}`);
  return phone.number; // E.164
}
Where does defaultCountry come from? Telegram contacts always include the country code, so for drivers it doesn't matter. When a tenant admin types a member's number in the dashboard, pass the tenant's country (stored on Tenant.country, an ISO code like IN or GB).

Seed

Creates: a super admin, one tenant + tenant admin, one hybrid charger CP_TEST01 (FLEET free + PUBLIC ₹18) with 2 connectors, one fleet member.

packages/db/prisma/seed.ts
import bcrypt from "bcryptjs";
import type { CountryCode } from "libphonenumber-js";
import { prisma } from "../src/client";
import { normalizePhone } from "../src/phone";

async function main() {
  const hash = await bcrypt.hash("admin123", 10);

  await prisma.adminUser.upsert({
    where: { email: "super@ev.local" }, update: {},
    create: { email: "super@ev.local", passwordHash: hash, role: "SUPER_ADMIN" },
  });

  const tenant = await prisma.tenant.upsert({
    where: { slug: "tata-fleet" }, update: {},
    create: { name: "Tata Fleet", slug: "tata-fleet", country: "IN" },
  });

  await prisma.adminUser.upsert({
    where: { email: "admin@tata.local" }, update: {},
    create: { email: "admin@tata.local", passwordHash: hash, role: "TENANT_ADMIN", tenantId: tenant.id },
  });

  const cp = await prisma.chargePoint.upsert({
    where: { ocppId: "CP_TEST01" }, update: {},
    create: {
      ocppId: "CP_TEST01", name: "Depot Bay 1", tenantId: tenant.id,
      allowedModes: ["FLEET", "PUBLIC"],
      connectors: { create: [{ connectorId: 1 }, { connectorId: 2 }] },
    },
  });

  for (const [mode, price] of [["FLEET", 0], ["PUBLIC", 18]] as const) {
    await prisma.chargerTariffRule.upsert({
      where: { chargePointId_mode: { chargePointId: cp.id, mode } }, update: {},
      create: { chargePointId: cp.id, tenantId: tenant.id, mode, pricePerKwh: price },
    });
  }

  // Put YOUR phone here (any format) to test fleet mode in Telegram. Always save phones normalized.
  const phone = normalizePhone("99999 99999", tenant.country as CountryCode);
  await prisma.tenantMembership.upsert({
    where: { tenantId_phone: { tenantId: tenant.id, phone } }, update: {},
    create: { tenantId: tenant.id, phone, role: "FLEET", name: "Test Driver" },
  });
}

main().finally(() => prisma.$disconnect());

Prove the Docker image builds (without Docker on your PC)

The schema now exists, so the Day 1 Dockerfile can build. GitHub builds it for you on every push, and also publishes it to GitHub's free container registry (GHCR), which is where your server will pull it from on Day 14.

Where the files go: your git repo root is D:\ocpp, and the app is in D:\ocpp\ev-platform. GitHub only reads workflows from the git root, so create D:\ocpp\.github\workflows\docker.yml (not inside ev-platform). The Dockerfile stays in ev-platform, which is why the workflow sets context: ./ev-platform.
.github/workflows/docker.yml
name: docker
on:
  push:
    branches: [main]
    paths: ["ev-platform/**", ".github/workflows/docker.yml"] # guide/ edits don't need a new image
  pull_request:
    paths: ["ev-platform/**", ".github/workflows/docker.yml"]

permissions:
  contents: read
  packages: write

jobs:
  api-image:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
      - uses: docker/setup-buildx-action@f87e5991a6d7451dcb8d9637bfbc97413f497069 # v4.4.1
      - uses: docker/login-action@81c6f93d4bc991e510b342a00b419b3bb4efc5c6 # v4.7.0
        if: github.event_name == 'push'
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}
      - uses: docker/build-push-action@c3c9e263c25d99ce0380d002d59b67737d91b0dc # v7.4.0
        with:
          context: ./ev-platform   # the app lives in this subfolder of the git repo
          push: ${{ github.event_name == 'push' }}
          tags: |
            ghcr.io/${{ github.repository_owner }}/ev-api:latest
            ghcr.io/${{ github.repository_owner }}/ev-api:${{ github.sha }}
          cache-from: type=gha
          cache-to: type=gha,mode=max
.github/dependabot.yml
version: 2
updates:
  - package-ecosystem: github-actions
    directory: /
    schedule:
      interval: weekly
  • Actions are pinned to commit SHAs (the version is in the comment). A tag like @v4 can be moved by the action's author, and this job can publish packages; a SHA can't change. The Dependabot file below opens a PR when a new version comes out. To find a SHA yourself: git ls-remote https://github.com/docker/login-action.git refs/tags/v4.7.0.
  • Pull requests only build the image (to catch Dockerfile mistakes). Pushes to main also publish it.
  • Docker image names must be lowercase. If your GitHub username has capitals, replace ${{ github.repository_owner }} with the lowercase name.
  • Image tagged with the commit SHA = you can roll back to any older version on the server.
  • If the build fails, read the red step in the Actions tab. The most common causes are a missing file in a COPY line or a lockfile that isn't committed.
