Day 13 — Tenant dashboard

Goal: a tenant admin manages only their own chargers, modes, tariffs, QR codes, members and sessions.

Pages

  • /dashboard/chargers — live status (poll every 10 s), mode checkboxes + ₹/kWh per mode, QR download per connector
  • /dashboard/members — add phone / name / vehicle / role, deactivate
  • /dashboard/sessions — filter by date & mode, totals for kWh and ₹

Tasks

apps/api/src/routes/tenant.ts
import { Router } from "express";
import { z } from "zod";
import { prisma, normalizePhone } from "@ev/db";
import { requireRole } from "../middleware/auth";

export const tenant = Router();
tenant.use(requireRole("TENANT_ADMIN"));

const Mode = z.enum(["PUBLIC", "PRIVATE_MEMBER", "FLEET"]);

// ── Chargers ─────────────────────────────────────────
tenant.get("/chargers", async (req, res) => {
  res.json(await req.db.chargePoint.findMany({
    include: { connectors: { orderBy: { connectorId: "asc" } }, tariffRules: true },
    orderBy: { ocppId: "asc" },
  }));
});

tenant.put("/chargers/:id/modes", async (req, res) => {
  const { modes } = z.object({
    modes: z.array(z.object({ mode: Mode, pricePerKwh: z.number().min(0).max(100) })),
  }).parse(req.body);

  // Ownership check through tenantDb, then plain prisma for the multi-step write.
  const cp = await req.db.chargePoint.findFirstOrThrow({ where: { id: req.params.id } });
  const list = modes.map((m) => m.mode);

  await prisma.$transaction([
    prisma.chargePoint.update({ where: { id: cp.id }, data: { allowedModes: list } }),
    prisma.chargerTariffRule.deleteMany({ where: { chargePointId: cp.id, mode: { notIn: list } } }),
    ...modes.map((m) => prisma.chargerTariffRule.upsert({
      where: { chargePointId_mode: { chargePointId: cp.id, mode: m.mode } },
      update: { pricePerKwh: m.pricePerKwh },
      create: { chargePointId: cp.id, tenantId: cp.tenantId, mode: m.mode, pricePerKwh: m.pricePerKwh },
    })),
  ]);
  res.json({ ok: true });
});

// ── Members ──────────────────────────────────────────
tenant.get("/members", async (req, res) => {
  res.json(await req.db.tenantMembership.findMany({ orderBy: { name: "asc" } }));
});

tenant.post("/members", async (req, res) => {
  const body = z.object({
    phone: z.string().min(10),
    name: z.string().optional(),
    vehicleNo: z.string().optional(),
    role: z.enum(["FLEET", "PRIVATE_MEMBER"]),
  }).parse(req.body);

  const phone = normalizePhone(body.phone);
  const existing = await prisma.user.findUnique({ where: { phone } }); // already onboarded? link now
  const m = await req.db.tenantMembership.create({
    data: { ...body, phone, userId: existing?.id ?? null } as any, // tenantId injected by tenantDb
  });
  res.status(201).json(m);
});

tenant.patch("/members/:id", async (req, res) => {
  const { isActive } = z.object({ isActive: z.boolean() }).parse(req.body);
  res.json(await req.db.tenantMembership.update({ where: { id: req.params.id }, data: { isActive } }));
});

// ── Sessions & revenue ───────────────────────────────
tenant.get("/sessions", async (req, res) => {
  const q = z.object({
    from: z.coerce.date().optional(),
    to: z.coerce.date().optional(),
    mode: Mode.optional(),
  }).parse(req.query);

  const where = { mode: q.mode, createdAt: { gte: q.from, lte: q.to } };
  const [rows, totals] = await Promise.all([
    req.db.chargingSession.findMany({
      where,
      include: {
        user: { select: { name: true, phone: true } },
        connector: { select: { connectorId: true, chargePoint: { select: { ocppId: true } } } },
      },
      orderBy: { createdAt: "desc" },
      take: 200,
    }),
    req.db.chargingSession.aggregate({ where: { ...where, status: "COMPLETED" }, _sum: { energyKwh: true, amount: true } }),
  ]);
  res.json({ rows, totals: totals._sum });
});

Mode & tariff editor (one per charger row)

apps/web/src/components/ModeEditor.tsx
"use client";
import { useState } from "react";
import { api } from "@/lib/api";

const MODES = ["FLEET", "PRIVATE_MEMBER", "PUBLIC"] as const;

export function ModeEditor({ cp, onSaved }: { cp: any; onSaved: () => void }) {
  const initial = Object.fromEntries(MODES.map((m) => {
    const rule = cp.tariffRules.find((r: any) => r.mode === m);
    return [m, { on: cp.allowedModes.includes(m), price: rule ? Number(rule.pricePerKwh) : 0 }];
  }));
  const [state, setState] = useState(initial);

  async function save() {
    const modes = MODES.filter((m) => state[m].on).map((m) => ({ mode: m, pricePerKwh: state[m].price }));
    await api(`/tenant/chargers/${cp.id}/modes`, { method: "PUT", body: JSON.stringify({ modes }) });
    onSaved();
  }

  return (
    <div className="flex items-center gap-4">
      {MODES.map((m) => (
        <label key={m} className="flex items-center gap-1">
          <input type="checkbox" checked={state[m].on}
            onChange={(e) => setState({ ...state, [m]: { ...state[m], on: e.target.checked } })} />
          {m} ₹
          <input type="number" min={0} step={0.5} value={state[m].price} disabled={!state[m].on}
            className="w-16 rounded border p-1"
            onChange={(e) => setState({ ...state, [m]: { ...state[m], price: Number(e.target.value) } })} />
        </label>
      ))}
      <button onClick={save} className="rounded bg-black px-3 py-1 text-white">Save</button>
    </div>
  );
}

QR download (one button per connector)

apps/web/src/components/QrButton.tsx
"use client";
import QRCode from "qrcode";

export function QrButton({ ocppId, connectorId }: { ocppId: string; connectorId: number }) {
  async function download() {
    const link = `https://t.me/${process.env.NEXT_PUBLIC_BOT_USERNAME}?start=cp_${ocppId}_conn_${connectorId}`;
    const png = await QRCode.toDataURL(link, { width: 800, margin: 2 });
    const a = document.createElement("a");
    a.href = png;
    a.download = `${ocppId}-connector-${connectorId}.png`;
    a.click();
  }
  return <button onClick={download} className="rounded border px-2 py-1">QR #{connectorId}</button>;
}
